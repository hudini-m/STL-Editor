# File: src/main_window.py
import sys
from pathlib import Path
from PySide6.QtWidgets import (
    QMainWindow, QMenuBar, QToolBar, QStatusBar,
    QWidget, QVBoxLayout, QHBoxLayout, QPushButton,
    QFileDialog, QLabel, QSplitter
)
from PySide6.QtCore import Qt
from viewport import ViewportWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("STL Mesh Editor")
        self.setGeometry(100, 100, 1200, 800)

        # Create central widget and layout
        central_widget = QWidget()
        self.setCentralWidget(central_widget)
        main_layout = QVBoxLayout(central_widget)

        # Create menu bar
        self.create_menu_bar()

        # Create toolbar
        self.create_toolbar()

        # Create status bar
        self.status_bar = QStatusBar()
        self.setStatusBar(self.status_bar)

        # Create main splitter for viewport and controls
        splitter = QSplitter(Qt.Horizontal)
        main_layout.addWidget(splitter)

        # Create 3D viewport
        self.viewport = ViewportWidget()
        splitter.addWidget(self.viewport)

        # Create right panel with editing controls
        from ui.editing_panel import EditingPanel
        self.editing_panel = EditingPanel()
        splitter.addWidget(self.editing_panel)

        # Set splitter sizes
        splitter.setSizes([800, 400])

        # Connect signals
        self.viewport.mesh_loaded.connect(self.on_mesh_loaded)

    def create_menu_bar(self):
        menubar = self.menuBar()

        # File menu
        file_menu = menubar.addMenu("File")
        open_action = file_menu.addAction("Open STL")
        save_action = file_menu.addAction("Save STL")
        exit_action = file_menu.addAction("Exit")

        # Edit menu
        edit_menu = menubar.addMenu("Edit")
        undo_action = edit_menu.addAction("Undo")
        redo_action = edit_menu.addAction("Redo")
        reset_action = edit_menu.addAction("Reset")

        # View menu
        view_menu = menubar.addMenu("View")
        fit_action = view_menu.addAction("Fit to View")
        front_action = view_menu.addAction("Front View")
        back_action = view_menu.addAction("Back View")
        left_action = view_menu.addAction("Left View")
        right_action = view_menu.addAction("Right View")
        top_action = view_menu.addAction("Top View")
        bottom_action = view_menu.addAction("Bottom View")
        perspective_action = view_menu.addAction("Perspective")
        orthographic_action = view_menu.addAction("Orthographic")
        wireframe_action = view_menu.addAction("Wireframe")

        # Connect actions
        open_action.triggered.connect(self.open_stl)
        save_action.triggered.connect(self.save_stl)
        exit_action.triggered.connect(self.close)
        undo_action.triggered.connect(self.undo)
        redo_action.triggered.connect(self.redo)
        reset_action.triggered.connect(self.reset)
        fit_action.triggered.connect(self.viewport.fit_to_view)
        front_action.triggered.connect(lambda: self.viewport.set_camera_view('front'))
        back_action.triggered.connect(lambda: self.viewport.set_camera_view('back'))
        left_action.triggered.connect(lambda: self.viewport.set_camera_view('left'))
        right_action.triggered.connect(lambda: self.viewport.set_camera_view('right'))
        top_action.triggered.connect(lambda: self.viewport.set_camera_view('top'))
        bottom_action.triggered.connect(lambda: self.viewport.set_camera_view('bottom'))
        perspective_action.triggered.connect(lambda: self.viewport.set_camera_mode('perspective'))
        orthographic_action.triggered.connect(lambda: self.viewport.set_camera_mode('orthographic'))
        wireframe_action.triggered.connect(self.viewport.toggle_wireframe)

    def create_toolbar(self):
        toolbar = QToolBar("Main Toolbar")
        self.addToolBar(toolbar)

        # Add buttons to toolbar
        open_btn = QPushButton("Open STL")
        save_btn = QPushButton("Save STL")
        undo_btn = QPushButton("Undo")
        redo_btn = QPushButton("Redo")
        reset_btn = QPushButton("Reset")

        toolbar.addWidget(open_btn)
        toolbar.addWidget(save_btn)
        toolbar.addWidget(undo_btn)
        toolbar.addWidget(redo_btn)
        toolbar.addWidget(reset_btn)

        # Connect buttons
        open_btn.clicked.connect(self.open_stl)
        save_btn.clicked.connect(self.save_stl)
        undo_btn.clicked.connect(self.undo)
        redo_btn.clicked.connect(self.redo)
        reset_btn.clicked.connect(self.reset)

    def open_stl(self):
        file_path, _ = QFileDialog.getOpenFileName(
            self, "Open STL File", "", "STL Files (*.stl);;All Files (*)"
        )
        if file_path:
            self.viewport.load_mesh(file_path)

    def save_stl(self):
        file_path, _ = QFileDialog.getSaveFileName(
            self, "Save STL File", "", "STL Files (*.stl);;All Files (*)"
        )
        if file_path:
            self.viewport.save_mesh(file_path)

    def undo(self):
        self.viewport.undo()

    def redo(self):
        self.viewport.redo()

    def reset(self):
        self.viewport.reset_view()

    def on_mesh_loaded(self, mesh_info):
        self.status_bar.showMessage(f"Loaded: {mesh_info['name']} - Vertices: {mesh_info['vertices']}, Faces:
{mesh_info['faces']}")

import sys
from pathlib import Path
from PySide6.QtWidgets import (
    QWidget, QVBoxLayout, QHBoxLayout, QPushButton,
    QSlider, QLabel, QFrame, QSplitter
)
from PySide6.QtCore import Qt, Signal, QTimer
import pyvista as pv
import numpy as np

# Add src to Python path for imports
sys.path.insert(0, str(Path(__file__).parent))

class ViewportWidget(QWidget):
    mesh_loaded = Signal(dict)

    def __init__(self):
        super().__init__()

        # Create layout
        layout = QVBoxLayout(self)

        # Create PyVista plotter
        self.plotter = pv.Plotter()
        self.plotter.background_color = 'white'
        self.plotter.enable_anti_aliasing()

        # Add PyVista widget to layout
        from pyvistaqt import BackgroundPlotter
        self.plotter_widget = BackgroundPlotter(parent=self, off_screen=False)
        layout.addWidget(self.plotter_widget)

        # Create control bar
        control_layout = QHBoxLayout()
        self.control_frame = QFrame()
        self.control_frame.setFrameStyle(QFrame.StyledPanel)
        self.control_frame.setLayout(control_layout)
        layout.addWidget(self.control_frame)

        # Add controls to control bar
        self.add_controls(control_layout)

        # Initialize state
        self.mesh = None
        self.original_mesh = None
        self.history = []
        self.history_index = -1

    def add_controls(self, layout):
        # View controls
        view_btns = [
            ("Fit", self.fit_to_view),
            ("Front", lambda: self.set_camera_view('front')),
            ("Back", lambda: self.set_camera_view('back')),
            ("Left", lambda: self.set_camera_view('left')),
            ("Right", lambda: self.set_camera_view('right')),
            ("Top", lambda: self.set_camera_view('top')),
            ("Bottom", lambda: self.set_camera_view('bottom')),
            ("Perspective", lambda: self.set_camera_mode('perspective')),
            ("Orthographic", lambda: self.set_camera_mode('orthographic')),
        ]

        for text, func in view_btns:
            btn = QPushButton(text)
            btn.clicked.connect(func)
            layout.addWidget(btn)

        # Transparency slider
        transparency_label = QLabel("Transparency")
        self.transparency_slider = QSlider(Qt.Horizontal)
        self.transparency_slider.setMinimum(0)
        self.transparency_slider.setMaximum(100)
        self.transparency_slider.setValue(50)
        self.transparency_slider.valueChanged.connect(self.update_transparency)

        layout.addWidget(transparency_label)
        layout.addWidget(self.transparency_slider)

        # Wireframe toggle
        wireframe_btn = QPushButton("Wireframe")
        wireframe_btn.setCheckable(True)
        wireframe_btn.clicked.connect(self.toggle_wireframe)
        layout.addWidget(wireframe_btn)

    def load_mesh(self, file_path):
        try:
            import trimesh
            mesh = trimesh.load(file_path)

            # Convert to PyVista mesh
            vertices = np.array(mesh.vertices)
            faces = np.array([f for f in mesh.faces])

            self.mesh = pv.PolyData(vertices, faces)
            self.original_mesh = self.mesh.copy()

            # Add mesh to plotter
            self.plotter.add_mesh(self.mesh, color='lightblue',
                                 smooth_shading=True,
                                 opacity=0.7)

            # Update camera
            self.fit_to_view()

            # Emit signal with mesh info
            self.mesh_loaded.emit({
                'name': Path(file_path).name,
                'vertices': len(vertices),
                'faces': len(faces)
            })

        except Exception as e:
            print(f"Error loading mesh: {e}")

    def save_mesh(self, file_path):
        if self.mesh is not None:
            try:
                import trimesh
                # Convert PyVista to trimesh
                vertices = np.array(self.mesh.points)
                faces = np.array(self.mesh.faces)

                # Create trimesh object
                mesh = trimesh.Trimesh(vertices=vertices, faces=faces)

                # Export to STL
                mesh.export(file_path)
            except Exception as e:
                print(f"Error saving mesh: {e}")

    def fit_to_view(self):
        if self.plotter.renderers:
            self.plotter.reset_camera()

    def set_camera_view(self, view):
        views = {
            'front': (0, 0, 1),
            'back': (0, 0, -1),
            'left': (-1, 0, 0),
            'right': (1, 0, 0),
            'top': (0, 1, 0),
            'bottom': (0, -1, 0)
        }

        if view in views:
            self.plotter.camera_position = views[view]

    def set_camera_mode(self, mode):
        if mode == 'perspective':
            self.plotter.enable_perspective()
        else:
            self.plotter.enable_orthographic()

    def toggle_wireframe(self):
        # This is a placeholder - actual implementation will be in the next phase
        pass

    def update_transparency(self, value):
        if self.mesh is not None:
            opacity = value / 100.0
            self.plotter.meshes[0].GetProperty().SetOpacity(opacity)

    def undo(self):
        # Placeholder for undo functionality - will be implemented in later phases
        pass

    def redo(self):
        # Placeholder for redo functionality - will be implemented in later phases
        pass

    def reset_view(self):
        self.plotter.reset_camera()

# File: src/geometry/mesh_model.py
import numpy as np
import trimesh
import pyvista as pv

class MeshModel:
    def __init__(self, vertices=None, faces=None):
        self.vertices = vertices if vertices is not None else np.array([])
        self.faces = faces if faces is not None else np.array([])
        self.original_vertices = vertices.copy() if vertices is not None else np.array([])
        self.mesh = None
        self.trimesh_obj = None

    def load_from_trimesh(self, trimesh_obj):
        """Load mesh from a trimesh object"""
        self.trimesh_obj = trimesh_obj
        self.vertices = np.array(trimesh_obj.vertices)
        self.faces = np.array([f for f in trimesh_obj.faces])
        self.original_vertices = self.vertices.copy()

    def load_from_pyvista(self, pyvista_mesh):
        """Load mesh from a PyVista mesh"""
        self.mesh = pyvista_mesh
        self.vertices = np.array(pyvista_mesh.points)
        self.faces = np.array(pyvista_mesh.faces)
        self.original_vertices = self.vertices.copy()

    def to_trimesh(self):
        """Convert back to trimesh object"""
        if self.trimesh_obj is not None:
            return self.trimesh_obj
        else:
            # Create new trimesh from our data
            return trimesh.Trimesh(vertices=self.vertices, faces=self.faces)

    def to_pyvista(self):
        """Convert back to PyVista mesh"""
        if self.mesh is not None:
            return self.mesh
        else:
            # Create new PyVista mesh from our data
            return pv.PolyData(self.vertices, self.faces)

    def get_vertex_count(self):
        """Get number of vertices"""
        return len(self.vertices)

    def get_face_count(self):
        """Get number of faces"""
        return len(self.faces)

    def get_bounds(self):
        """Get bounding box of mesh"""
        if len(self.vertices) == 0:
            return None
        return np.array([
            [self.vertices[:, i].min() for i in range(3)],
            [self.vertices[:, i].max() for i in range(3)]
        ])

    def get_center(self):
        """Get center point of mesh"""
        bounds = self.get_bounds()
        if bounds is None:
            return np.array([0, 0, 0])
        return (bounds[0] + bounds[1]) / 2.0

    def copy(self):
        """Create a deep copy of the mesh model"""
        new_model = MeshModel(self.vertices.copy(), self.faces.copy())
        new_model.original_vertices = self.original_vertices.copy()
        return new_model

# File: src/interaction/picking.py
import numpy as np

class PickingManager:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model

    def pick_vertex(self, point, max_distance=1.0):
        """Pick a vertex near the given point"""
        if len(self.mesh_model.vertices) == 0:
            return None

        # Calculate distances from all vertices to the point
        distances = np.linalg.norm(
            self.mesh_model.vertices - np.array(point),
            axis=1
        )

        # Find closest vertex within max_distance
        min_idx = np.argmin(distances)
        if distances[min_idx] <= max_distance:
            return min_idx
        return None

    def pick_face(self, point):
        """Pick a face near the given point"""
        # This is a simplified version - would require more complex intersection
        # detection in practice (using trimesh or similar libraries)
        if len(self.mesh_model.vertices) == 0:
            return None

        # For now, just return closest vertex to simplify
        distances = np.linalg.norm(
            self.mesh_model.vertices - np.array(point),
            axis=1
        )
        min_idx = np.argmin(distances)
        return min_idx

    def get_nearest_face(self, point):
        """Get the face nearest to a point (simplified implementation)"""
        # Simplified: return face containing the closest vertex
        vertex_idx = self.pick_vertex(point)
        if vertex_idx is None:
            return None

        # Find faces that contain this vertex
        faces = []
        for i, face in enumerate(self.mesh_model.faces):
            if vertex_idx in face:
                faces.append(i)

        return faces[0] if faces else None

import numpy as np

class DeformationEngine:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model

    def compute_distance_weights(self, control_points, vertices, influence_radius, falloff_type='smoothstep'):
        """
        Compute weights for each vertex based on distance to control points.

        Args:
            control_points: array of control point positions
            vertices: array of vertex positions
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')

        Returns:
            weights: array of weights for each vertex (0.0 to 1.0)
        """
        if len(control_points) == 0 or len(vertices) == 0:
            return np.zeros(len(vertices))

        # Compute distances from all vertices to all control points
        # Shape: (num_vertices, num_control_points)
        distances = np.linalg.norm(
            vertices[:, np.newaxis, :] - control_points[np.newaxis, :, :],
            axis=2
        )

        # Find minimum distance for each vertex to any control point
        min_distances = np.min(distances, axis=1)

        # Compute weights based on influence radius and falloff type
        weights = np.zeros(len(vertices))

        # Only consider vertices within influence radius
        within_radius = min_distances <= influence_radius

        if not np.any(within_radius):
            return weights

        # Normalize distances to [0, 1] range for the falloff function
        normalized_distances = min_distances[within_radius] / influence_radius

        if falloff_type == 'linear':
            # Linear falloff: weight decreases linearly from 1.0 to 0.0
            weights[within_radius] = 1.0 - normalized_distances

        elif falloff_type == 'smoothstep':
            # Smoothstep falloff: smooth S-shaped curve
            # https://en.wikipedia.org/wiki/Smoothstep
            t = np.clip(normalized_distances, 0.0, 1.0)
            weights[within_radius] = t * t * (3 - 2 * t)

        elif falloff_type == 'gaussian':
            # Gaussian falloff: weight follows a Gaussian distribution
            sigma = influence_radius / 3.0  # Adjust for desired spread
            weights[within_radius] = np.exp(-0.5 * (normalized_distances ** 2) / (sigma ** 2))

        return weights

    def apply_deformation(self, control_points, displacements,
                         influence_radius=1.0, falloff_type='smoothstep'):
        """
        Apply deformation to the mesh based on control points and displacements.

        Args:
            control_points: array of control point positions
            displacements: array of displacement vectors for each control point
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')

        Returns:
            deformed_vertices: array of new vertex positions
        """
        if len(self.mesh_model.vertices) == 0:
            return self.mesh_model.vertices

        # Compute weights for each vertex
        weights = self.compute_distance_weights(
            control_points,
            self.mesh_model.vertices,
            influence_radius,
            falloff_type
        )

        # Apply deformation to vertices
        deformed_vertices = self.mesh_model.vertices.copy()

        # For each vertex, compute weighted displacement
        for i, (vertex_pos, weight) in enumerate(zip(self.mesh_model.vertices, weights)):
            if weight > 0:
                # Apply weighted average of control point displacements
                total_displacement = np.zeros(3)
                for cp_pos, disp in zip(control_points, displacements):
                    dist = np.linalg.norm(vertex_pos - cp_pos)
                    if dist <= influence_radius:
                        # Weight by distance (inverse relationship)
                        local_weight = 1.0 - (dist / influence_radius)
                        total_displacement += disp * local_weight

                deformed_vertices[i] = vertex_pos + total_displacement * weight

        return deformed_vertices

    def apply_smooth_deformation(self, control_points, displacements,
                                influence_radius=1.0, falloff_type='smoothstep'):
        """
        Apply smooth deformation using the laplacian method (simplified).
        This is a more advanced version that maintains mesh quality.
        """
        # Simplified implementation - in practice would use proper laplacian
        # or ARAP deformation methods

        # For now, just apply the basic deformation with smoothing
        return self.apply_deformation(
            control_points, displacements, influence_radius, falloff_type
        )

        # For each vertex, compute how much it faces the camera
        # This is a simplified approach - in practice would use proper silhouette extraction

        # We'll simply return a subset of vertices based on their position
        # This needs to be enhanced for real contour detection

        # Return first 10% of vertices (simplified)
        num_contour_points = max(5, int(len(self.mesh_model.vertices) * 0.1))

        # Get the vertices with highest z values (assumed to be part of silhouette)
        z_values = self.mesh_model.vertices[:, 2]
        sorted_indices = np.argsort(z_values)[::-1]  # Descending order

        return sorted_indices[:num_contour_points].tolist()

    def sample_contour(self, contour_vertices, density=10):
        """
        Sample points along the contour at specified density.

        Args:
            contour_vertices: list of vertex indices forming the contour
            density: number of points to sample per unit length

        Returns:
            sampled_points: array of sampled point positions
        """
        if len(contour_vertices) < 2:
            return np.array([])

        # For simplicity, just return a subset of contour vertices
        step = max(1, len(contour_vertices) // density)
        sampled_indices = contour_vertices[::step]

        return self.mesh_model.vertices[sampled_indices]

    def generate_uniform_contour_points(self, num_points=10):
        """
        Generate uniformly distributed control points along the mesh surface.

        Args:
            num_points: number of points to generate

        Returns:
            points: array of point positions
        """
        if len(self.mesh_model.vertices) == 0:
            return np.array([])

        # Sample points from vertices uniformly
        indices = np.random.choice(len(self.mesh_model.vertices),
                                  min(num_points, len(self.mesh_model.vertices)),
                                  replace=False)

        return self.mesh_model.vertices[indices]

import numpy as np

class ControlPointManager:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model
        self.control_points = []  # List of (position, is_pinned) tuples
        self.selected_point = None

    def add_control_point(self, position, pinned=False):
        """Add a control point at the specified position"""
        self.control_points.append((np.array(position), pinned))

    def remove_control_point(self, index):
        """Remove a control point by index"""
        if 0 <= index < len(self.control_points):
            del self.control_points[index]

    def clear_all_points(self):
        """Remove all control points"""
        self.control_points = []

    def get_control_points(self):
        """Get list of control points positions"""
        return [cp[0] for cp in self.control_points]

    def get_pinned_points(self):
        """Get indices of pinned control points"""
        return [i for i, (pos, pinned) in enumerate(self.control_points) if pinned]

    def set_point_position(self, index, new_position):
        """Move a control point to a new position"""
        if 0 <= index < len(self.control_points):
            old_pos, pinned = self.control_points[index]
            self.control_points[index] = (np.array(new_position), pinned)

    def select_point(self, index):
        """Select a control point by index"""
        self.selected_point = index

    def get_selected_point(self):
        """Get the currently selected point index"""
        return self.selected_point

    def is_point_pinned(self, index):
        """Check if a point is pinned"""
        if 0 <= index < len(self.control_points):
            return self.control_points[index][1]
        return False

    def pin_point(self, index):
        """Pin a control point (prevent movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, True)

    def unpin_point(self, index):
        """Unpin a control point (allow movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, False)

    def get_count(self):
        """Get number of control points"""
        return len(self.control_points)

class ControlPointManager:

        if 0 <= index < len(self.control_points):
            del self.control_points[index]

    def clear_all_points(self):
        """Remove all control points"""
        self.control_points = []

    def get_control_points(self):
        """Get list of control point positions"""
        return [cp[0] for cp in self.control_points]

    def get_pinned_points(self):
        """Get indices of pinned control points"""
        return [i for i, (pos, pinned) in enumerate(self.control_points) if pinned]

    def set_point_position(self, index, new_position):
        """Move a control point to a new position"""
        if 0 <= index < len(self.control_points):
            old_pos, pinned = self.control_points[index]
            self.control_points[index] = (np.array(new_position), pinned)

    def select_point(self, index):
        """Select a control point by index"""
        if not self.is_point_pinned(index):  # Don't select pinned points
            self.selected_point = index

    def get_selected_point(self):
        """Get the currently selected point index"""
        return self.selected_point

    def is_point_pinned(self, index):
        """Check if a point is pinned"""
        if 0 <= index < len(self.control_points):
            return self.control_points[index][1]
        return False

    def pin_point(self, index):
        """Pin a control point (prevent movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, True)

    def unpin_point(self, index):
        """Unpin a control point (allow movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, False)

    def get_count(self):
        """Get number of control points"""
        return len(self.control_points)

    def start_dragging(self, point_index):
        """Start dragging a control point"""
        if 0 <= point_index < len(self.control_points) and not self.is_point_pinned(point_index):
            self.dragging_point = point_index

    def stop_dragging(self):
        """Stop dragging control points"""
        self.dragging_point = None

        """Get the index of currently dragging point"""

# File: src/geometry/deformation.py (updated with better interface)
import numpy as np

class DeformationEngine:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model

    def compute_distance_weights(self, control_points, vertices, influence_radius, falloff_type='smoothstep'):
        """
        Compute weights for each vertex based on distance to control points.

        Args:
            control_points: array of control point positions
            vertices: array of vertex positions
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')

        Returns:
            weights: array of weights for each vertex (0.0 to 1.0)
        """
        if len(control_points) == 0 or len(vertices) == 0:
            return np.zeros(len(vertices))

        # Compute distances from all vertices to all control points
        # Shape: (num_vertices, num_control_points)
        distances = np.linalg.norm(
            vertices[:, np.newaxis, :] - control_points[np.newaxis, :, :],
            axis=2
        )

        # Find minimum distance for each vertex to any control point
        min_distances = np.min(distances, axis=1)

        # Compute weights based on influence radius and falloff type
        weights = np.zeros(len(vertices))

        # Only consider vertices within influence radius
        within_radius = min_distances <= influence_radius

        if not np.any(within_radius):
            return weights

        # Normalize distances to [0, 1] range for the falloff function
        normalized_distances = min_distances[within_radius] / influence_radius

        if falloff_type == 'linear':
            # Linear falloff: weight decreases linearly from 1.0 to 0.0
            weights[within_radius] = 1.0 - normalized_distances

        elif falloff_type == 'smoothstep':
            # Smoothstep falloff: smooth S-shaped curve
            # https://en.wikipedia.org/wiki/Smoothstep
            t = np.clip(normalized_distances, 0.0, 1.0)
            weights[within_radius] = t * t * (3 - 2 * t)

        elif falloff_type == 'gaussian':
            # Gaussian falloff: weight follows a Gaussian distribution
            sigma = influence_radius / 3.0  # Adjust for desired spread
            weights[within_radius] = np.exp(-0.5 * (normalized_distances ** 2) / (sigma ** 2))

        return weights

    def apply_deformation(self, control_points, displacements,
                         influence_radius=1.0, falloff_type='smoothstep',
                         strength=1.0):
        """
        Apply deformation to the mesh based on control points and displacements.

        Args:
            control_points: array of control point positions
            displacements: array of displacement vectors for each control point
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')
            strength: overall strength of deformation (0.0 to 1.0)

        Returns:
            deformed_vertices: array of new vertex positions
        """
        if len(self.mesh_model.vertices) == 0:
            return self.mesh_model.vertices

        # Compute weights for each vertex
        weights = self.compute_distance_weights(
            control_points,
            self.mesh_model.vertices,
            influence_radius,
            falloff_type
        )

        # Apply deformation to vertices
        deformed_vertices = self.mesh_model.vertices.copy()

        # For each vertex, compute weighted displacement
        for i, (vertex_pos, weight) in enumerate(zip(self.mesh_model.vertices, weights)):
            if weight > 0:
                # Apply weighted average of control point displacements
                total_displacement = np.zeros(3)

                # Compute influence of each control point on this vertex
                for cp_pos, disp in zip(control_points, displacements):
                    dist = np.linalg.norm(vertex_pos - cp_pos)
                    if dist <= influence_radius:
                        # Weight by distance (inverse relationship)
                        local_weight = 1.0 - (dist / influence_radius)
                        total_displacement += disp * local_weight

                deformed_vertices[i] = vertex_pos + total_displacement * weight * strength

        return deformed_vertices

    def apply_smooth_deformation(self, control_points, displacements,
                                influence_radius=1.0, falloff_type='smoothstep',
                                strength=1.0):
        """
        Apply smooth deformation using the laplacian method (simplified).
        This is a more advanced version that maintains mesh quality.
        """
        # Simplified implementation - in practice would use proper laplacian
        # or ARAP deformation methods

        # For now, just apply the basic deformation with smoothing
        return self.apply_deformation(
            control_points, displacements, influence_radius, falloff_type, strength
        )

    def compute_vertex_displacement(self, vertex_idx, control_points, displacements,
                                   influence_radius=1.0, falloff_type='smoothstep'):
        """
        Compute displacement for a single vertex based on its distance to control points.

        Returns:
            displacement_vector: displacement vector for the vertex
        """
        if len(control_points) == 0:
            return np.array([0.0, 0.0, 0.0])

        # Find all control points that influence this vertex
        total_displacement = np.zeros(3)

        for cp_pos, disp in zip(control_points, displacements):
            dist = np.linalg.norm(self.mesh_model.vertices[vertex_idx] - cp_pos)
            if dist <= influence_radius:
                # Weight by distance (inverse relationship)
                local_weight = 1.0 - (dist / influence_radius)
                total_displacement += disp * local_weight

        return total_displacement

# File: src/geometry/contour.py (updated)
import numpy as np

class ContourExtractor:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model

    def extract_contour(self, view_direction=None, threshold=0.1):
        """
        Extract the silhouette or contour of the mesh.

        Args:
            view_direction: direction vector for viewing (if None, uses default)
            threshold: threshold for silhouette detection

        Returns:
            contour_vertices: array of contour vertex indices
        """
        if len(self.mesh_model.vertices) == 0:
            return []

        # If no view direction specified, use a default direction
        if view_direction is None:
            view_direction = np.array([0, 0, 1])

        # Normalize the view direction
        view_direction = view_direction / np.linalg.norm(view_direction)

        # For each vertex, compute how much it faces the camera
        # This is a simplified approach - in practice would use proper silhouette extraction

        # Project vertices onto the view direction to determine visibility
        projections = np.dot(self.mesh_model.vertices, view_direction)

        # Find the maximum and minimum projections for range
        max_proj = np.max(projections)
        min_proj = np.min(projections)

        # Define a threshold based on the range of projections
        range_proj = max_proj - min_proj

        # Get vertices that are on the silhouette (at edges)
        # This is a simple approximation - in practice would use more robust methods
        contour_threshold = min_proj + range_proj * threshold

        # Find all vertices with projection near the edge
        contour_indices = np.where(projections >= contour_threshold)[0]

        # Return unique indices, but also include some from the opposite end
        if len(contour_indices) < 10:
            # If not enough points found, add more from the other side
            other_indices = np.where(projections <= min_proj + range_proj * (1 - threshold))[0]
            contour_indices = np.concatenate([contour_indices, other_indices])

        # Return a subset of these indices to make it manageable
        if len(contour_indices) > 20:
            contour_indices = np.random.choice(contour_indices, size=20, replace=False)

        return contour_indices.tolist()

    def sample_contour(self, contour_vertices, density=10):
        """
        Sample points along the contour at specified density.

        Args:
            contour_vertices: list of vertex indices forming the contour
            density: number of points to sample per unit length

        Returns:
            sampled_points: array of sampled point positions
        """
        if len(contour_vertices) < 2:
            return np.array([])

        # Get actual vertex positions for sampling
        vertex_positions = self.mesh_model.vertices[contour_vertices]

        # For simplicity, just return a subset of contour vertices
        step = max(1, len(contour_vertices) // density)
        sampled_indices = contour_vertices[::step]

        return self.mesh_model.vertices[sampled_indices]

    def generate_uniform_contour_points(self, num_points=10):
        """
        Generate uniformly distributed control points along the mesh surface.

        Args:
            num_points: number of points to generate

        Returns:
            points: array of point positions
        """
        if len(self.mesh_model.vertices) == 0:
            return np.array([])

        # Sample points from vertices uniformly
        indices = np.random.choice(len(self.mesh_model.vertices),
                                  min(num_points, len(self.mesh_model.vertices)),
                                  replace=False)

        return self.mesh_model.vertices[indices]

    def get_mesh_edges(self):
        """
        Get all edges of the mesh.

        Returns:
            edges: list of (vertex1, vertex2) tuples
        """
        if len(self.mesh_model.faces) == 0:
            return []

        # Extract unique edges from faces
        edges = set()
        for face in self.mesh_model.faces:
            for i in range(len(face)):
                edge = tuple(sorted([face[i], face[(i+1) % len(face)]]))
                edges.add(edge)

        return list(edges)

    def get_boundary_vertices(self):
        """
        Get vertices that are on the boundary of the mesh.

        Returns:
            boundary_vertices: array of vertex indices
        """
        if len(self.mesh_model.faces) == 0:
            return []

        # Find all edges and count how many faces use each edge
        edges = self.get_mesh_edges()

        # Edges used by only one face are boundary edges
        edge_counts = {}
        for face in self.mesh_model.faces:
            for i in range(len(face)):
                edge = tuple(sorted([face[i], face[(i+1) % len(face)]]))
                if edge not in edge_counts:
                    edge_counts[edge] = 0
                edge_counts[edge] += 1

        # Get vertices that are on boundary edges
        boundary_vertices = set()
        for edge, count in edge_counts.items():
            if count == 1:  # Boundary edge
                boundary_vertices.add(edge[0])
                boundary_vertices.add(edge[1])

        return list(boundary_vertices)

import numpy as np

class ControlPointManager:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model
        self.control_points = []  # List of (position, is_pinned) tuples
        self.selected_point = None
        self.dragging_point = None

    def add_control_point(self, position, pinned=False):
        """Add a control point at the specified position"""
        self.control_points.append((np.array(position), pinned))

    def remove_control_point(self, index):
        """Remove a control point by index"""
        if 0 <= index < len(self.control_points):
            del self.control_points[index]

    def clear_all_points(self):
        """Remove all control points"""
        self.control_points = []

    def get_control_points(self):
        """Get list of control point positions"""
        return [cp[0] for cp in self.control_points]

    def get_pinned_points(self):
        """Get indices of pinned control points"""
        return [i for i, (pos, pinned) in enumerate(self.control_points) if pinned]

    def set_point_position(self, index, new_position):
        """Move a control point to a new position"""
        if 0 <= index < len(self.control_points):
            old_pos, pinned = self.control_points[index]
            self.control_points[index] = (np.array(new_position), pinned)

    def select_point(self, index):
        """Select a control point by index"""
        if not self.is_point_pinned(index):  # Don't select pinned points
            self.selected_point = index

    def get_selected_point(self):
        """Get the currently selected point index"""
        return self.selected_point

    def is_point_pinned(self, index):
        """Check if a point is pinned"""
        if 0 <= index < len(self.control_points):
            return self.control_points[index][1]
        return False

    def pin_point(self, index):
        """Pin a control point (prevent movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, True)

    def unpin_point(self, index):
        """Unpin a control point (allow movement)"""
        if 0 <= index < len(self.control_points):
            pos, _ = self.control_points[index]
            self.control_points[index] = (pos, False)

    def get_count(self):
        """Get number of control points"""
        return len(self.control_points)

    def start_dragging(self, point_index):
        """Start dragging a control point"""
        if 0 <= point_index < len(self.control_points) and not self.is_point_pinned(point_index):
            self.dragging_point = point_index

    def stop_dragging(self):
        """Stop dragging control points"""
        self.dragging_point = None

    def get_dragging_point(self):
        """Get the index of currently dragging point"""
        return self.dragging_point

    def update_dragged_point_position(self, new_position):
        """Update position of dragged point"""
        if self.dragging_point is not None:
            pos, pinned = self.control_points[self.dragging_point]
            self.control_points[self.dragging_point] = (np.array(new_position), pinned)

    def auto_generate_contour_points(self, density=10, view_direction=None):
        """
        Automatically generate control points along the mesh contour.

        Args:
            density: number of points to generate per unit length
            view_direction: direction vector for viewing

        Returns:
            num_generated: number of points generated
        """
        from geometry.contour import ContourExtractor

        if self.mesh_model is None:
            return 0

        extractor = ContourExtractor(self.mesh_model)

        # Extract contour vertices
        contour_indices = extractor.extract_contour(view_direction)

        # Sample points along the contour
        sampled_points = extractor.sample_contour(contour_indices, density)

        # Add each sampled point as a control point
        for point in sampled_points:
            self.add_control_point(point)

        return len(sampled_points)

    def generate_uniform_points(self, num_points=10):
        """
        Generate uniformly distributed control points.

        Args:
            num_points: number of points to generate

        Returns:
            num_generated: number of points generated
        """
        from geometry.contour import ContourExtractor

        if self.mesh_model is None:
            return 0

        extractor = ContourExtractor(self.mesh_model)
        points = extractor.generate_uniform_contour_points(num_points)

        # Add each point as a control point
        for point in points:
            self.add_control_point(point)

        return len(points)

# File: src/geometry/deformation.py (updated with pinned point handling)
import numpy as np

class DeformationEngine:
    def __init__(self, mesh_model):
        self.mesh_model = mesh_model

    def compute_distance_weights(self, control_points, vertices, influence_radius, falloff_type='smoothstep'):
        """
        Compute weights for each vertex based on distance to control points.

        Args:
            control_points: array of control point positions
            vertices: array of vertex positions
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')

        Returns:
            weights: array of weights for each vertex (0.0 to 1.0)
        """
        if len(control_points) == 0 or len(vertices) == 0:
            return np.zeros(len(vertices))

        # Compute distances from all vertices to all control points
        # Shape: (num_vertices, num_control_points)
        distances = np.linalg.norm(
            vertices[:, np.newaxis, :] - control_points[np.newaxis, :, :],
            axis=2
        )

        # Find minimum distance for each vertex to any control point
        min_distances = np.min(distances, axis=1)

        # Compute weights based on influence radius and falloff type
        weights = np.zeros(len(vertices))

        # Only consider vertices within influence radius
        within_radius = min_distances <= influence_radius

        if not np.any(within_radius):
            return weights

        # Normalize distances to [0, 1] range for the falloff function
        normalized_distances = min_distances[within_radius] / influence_radius

        if falloff_type == 'linear':
            # Linear falloff: weight decreases linearly from 1.0 to 0.0
            weights[within_radius] = 1.0 - normalized_distances

        elif falloff_type == 'smoothstep':
            # Smoothstep falloff: smooth S-shaped curve
            # https://en.wikipedia.org/wiki/Smoothstep
            t = np.clip(normalized_distances, 0.0, 1.0)
            weights[within_radius] = t * t * (3 - 2 * t)

        elif falloff_type == 'gaussian':
            # Gaussian falloff: weight follows a Gaussian distribution
            sigma = influence_radius / 3.0  # Adjust for desired spread
            weights[within_radius] = np.exp(-0.5 * (normalized_distances ** 2) / (sigma ** 2))

        return weights

    def apply_deformation(self, control_points, displacements,
                         influence_radius=1.0, falloff_type='smoothstep',
                         strength=1.0, pinned_indices=None):
        """
        Apply deformation to the mesh based on control points and displacements.

        Args:
            control_points: array of control point positions
            displacements: array of displacement vectors for each control point
            influence_radius: maximum distance for influence
            falloff_type: type of falloff ('linear', 'smoothstep', 'gaussian')
            strength: overall strength of deformation (0.0 to 1.0)
            pinned_indices: indices of control points that should not be moved

        Returns:
            deformed_vertices: array of new vertex positions
        """
        if len(self.mesh_model.vertices) == 0:
            return self.mesh_model.vertices

        # If no pinned indices specified, treat all as movable
        if pinned_indices is None:
            pinned_indices = []

        # Compute weights for each vertex
        weights = self.compute_distance_weights(
            control_points,
            self.mesh_model.vertices,
            influence_radius,
            falloff_type
        )

        # Apply deformation to vertices
        deformed_vertices = self.mesh_model.vertices.copy()

        # For each vertex, compute weighted displacement
        for i, (vertex_pos, weight) in enumerate(zip(self.mesh_model.vertices, weights)):
            if weight > 0:
                # Apply weighted average of control point displacements
                total_displacement = np.zeros(3)

                # Compute influence of each control point on this vertex
                for cp_idx, (cp_pos, disp) in enumerate(zip(control_points, displacements)):
                    dist = np.linalg.norm(vertex_pos - cp_pos)
                    if dist <= influence_radius:
                        # Weight by distance (inverse relationship)
                        local_weight = 1.0 - (dist / influence_radius)

                        # Don't apply displacement to pinned points
                        if cp_idx not in pinned_indices:
                            total_displacement += disp * local_weight

                deformed_vertices[i] = vertex_pos + total_displacement * weight * strength

        return deformed_vertices

    def apply_smooth_deformation(self, control_points, displacements,
                                influence_radius=1.0, falloff_type='smoothstep',
                                strength=1.0):
        """
        Apply smooth deformation using the laplacian method (simplified).
        This is a more advanced version that maintains mesh quality.
        """
        # Simplified implementation - in practice would use proper laplacian
        # or ARAP deformation methods

        # For now, just apply the basic deformation with smoothing
        return self.apply_deformation(
            control_points, displacements, influence_radius, falloff_type, strength
        )

    def compute_vertex_displacement(self, vertex_idx, control_points, displacements,
                                   influence_radius=1.0, falloff_type='smoothstep'):
        """
        Compute displacement for a single vertex based on its distance to control points.

        Returns:
            displacement_vector: displacement vector for the vertex
        """
        if len(control_points) == 0:
            return np.array([0.0, 0.0, 0.0])

        # Find all control points that influence this vertex
        total_displacement = np.zeros(3)

        for cp_pos, disp in zip(control_points, displacements):
            dist = np.linalg.norm(self.mesh_model.vertices[vertex_idx] - cp_pos)
            if dist <= influence_radius:
                # Weight by distance (inverse relationship)
                local_weight = 1.0 - (dist / influence_radius)
                total_displacement += disp * local_weight

        return total_displacement
